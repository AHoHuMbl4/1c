=== ubuntu/serenedb/serenedb_backup.sh (целиком, 28 строк) ===
     1	#!/usr/bin/env bash
     2	# Оффлайн-снапшот витрины SereneDB (проверено round-trip: снапшот → потеря → restore → данные на месте).
     3	# Кратко останавливает serenedb для консистентности. Ротация: держим последние KEEP.
     4	#
     5	# ⚠ ПЕРВИЧНОЕ восстановление витрины — РЕ-СИНК из 1С (`serene_sync`, витрина производная — не система
     6	#   оф-рекорд). Снапшот — БЫСТРЫЙ restore без повторного пула OData и ре-эмбеддинга резолвера.
     7	#
     8	# Запуск: bash serenedb_backup.sh   (или systemd-таймер, если нужен регулярный).
     9	set -euo pipefail
    10	DIR=${BACKUP_DIR:-/root/backups/serenedb}
    11	KEEP=${KEEP:-7}
    12	TS=$(date +%Y%m%d-%H%M%S)
    13	mkdir -p "$DIR"
    14
    15	systemctl stop serenedb
    16	tar czf "$DIR/serenedb-$TS.tar.gz" -C /var/lib/serenedb .
    17	systemctl start serenedb
    18
    19	# ротация
    20	ls -1t "$DIR"/serenedb-*.tar.gz 2>/dev/null | tail -n +$((KEEP + 1)) | xargs -r rm -f
    21	echo "backup: $DIR/serenedb-$TS.tar.gz ($(du -h "$DIR/serenedb-$TS.tar.gz" | cut -f1))"
    22
    23	# RESTORE (вручную):
    24	#   systemctl stop serenedb
    25	#   rm -rf /var/lib/serenedb/*
    26	#   tar xzf <backup.tar.gz> -C /var/lib/serenedb
    27	#   chown -R serenedb:serenedb /var/lib/serenedb
    28	#   systemctl start serenedb
